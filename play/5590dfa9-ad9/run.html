I've built the galaxy slime toy. It's a single file, `galaxy-slime.html`, with nothing to install and no internet needed. Double-click it to open it in Chrome, Edge or Firefox.

**How to play**
- **Drag** the slime to stretch it. A slow pull drags the whole blob along; a fast yank stretches it out and then tears a piece off with a "pop" sound.
- **Push pieces together** and they merge back into one blob with a "blorp" sound. A counter in the corner shows how many blobs there are.
- **Click empty space** and drag into the slime to poke and squish it.
- **Right-click** to pin a spot down, like a second hand holding it. Then pull the other side to stretch a long string before it snaps. Right-click the pin again to remove it. On a touchscreen you can just use two fingers.

**Buttons** (each also has a keyboard shortcut):
- ✨ **New slime** (R) starts over with a fresh blob.
- 🤲 **Squish together** (S) gathers all the bits back into the middle.
- 🎨 **Change galaxy** (C) switches between four colour schemes: Galaxy, Nebula, Supernova and Aurora.
- 🔬 **X-ray** (X) shows the slime as about 700 particles joined by stretchy bonds, with a short explanation of cohesion and bonds snapping.
- 🔊 **Sound** (M) turns the squelches and pops on or off.
- ⛶ **Full screen** (F).

**How I checked it**
- I ran the physics on its own and tested resting, stretching, tearing at different speeds, and gathering the pieces back together. It stayed stable throughout.
- I took screenshots in a headless browser of the normal view, X-ray mode, and a pinned stretch.
- I haven't played it with a real mouse or on a laptop. Try it on the laptop you'll use at the stall before the night.

**Tips for the stall**
- Use full screen and plug the laptop in. The toy runs a smooth animation constantly, so it will drain the battery.
- The usage hint at the bottom fades once someone starts playing and comes back after 25 seconds of nobody touching it.

'use strict';

// ===== SIM-START =====
// Particle slime: "viscoelastic fluid" model (Clavet, Beaudoin & Poulin 2005):
// double density relaxation (cohesion + incompressibility), plastic springs
// (stretchiness) and viscosity (gooeyness). All lengths in CSS pixels.
function createSim(W, H, h) {
  const MAXN = 4000;
  const px = new Float32Array(MAXN), py = new Float32Array(MAXN);
  const vx = new Float32Array(MAXN), vy = new Float32Array(MAXN);
  const qx = new Float32Array(MAXN), qy = new Float32Array(MAXN);
  const tu = new Float32Array(MAXN), tv = new Float32Array(MAXN);
  const spark = new Float32Array(MAXN);
  const rho = new Float32Array(MAXN), rhoN = new Float32Array(MAXN);
  const ddx = new Float32Array(MAXN), ddy = new Float32Array(MAXN);
  const held = new Uint8Array(MAXN);
  const cellOf = new Int32Array(MAXN), sorted = new Int32Array(MAXN);
  const uf = new Int32Array(MAXN), ufSize = new Int32Array(MAXN);
  const PAIRCAP = MAXN * 40;
  const pairI = new Int32Array(PAIRCAP), pairJ = new Int32Array(PAIRCAP);
  let cellStart = new Int32Array(1), cellCount = new Int32Array(1);
  const springs = new Map();   // key (i<<12)|j  ->  rest length
  const grabs = new Map();     // id -> {x, y, pin, idx, ox, oy}
  const pokers = new Map();    // id -> {x, y, r}

  const S = {
    n: 0, W, H, h, np: 0, gather: 0,
    px, py, vx, vy, tu, tv, spark, held, springs, grabs, pokers, pairI, pairJ,
    spacing: 0.5 * h,
    K: 0.30,        // pressure stiffness
    KN: 0.60,       // near-pressure stiffness (stops clumping)
    SIGMA: 0.15,    // linear viscosity
    BETA: 0.3 / h,  // quadratic viscosity
    KS: 0.80,       // spring stiffness
    GAMMA: 0.50,    // spring yield ratio
    ALPHA: 0.02,    // plasticity
    BREAK: h,       // springs snap once their rest length exceeds h (must not exceed h)
    FRICTION: 0.965,
    MAXV: 0.45 * h,
  };

  // Rest density = density of an interior particle in the starting hex lattice,
  // so a fresh blob starts out (almost) at rest.
  {
    const s = S.spacing; let r0 = 0;
    for (let a = -6; a <= 6; a++) for (let b = -6; b <= 6; b++) {
      if (!a && !b) continue;
      const x = (a + b * 0.5) * s, y = b * s * Math.sqrt(3) / 2, q = Math.hypot(x, y) / h;
      if (q < 1) r0 += (1 - q) * (1 - q);
    }
    S.R0 = r0;
  }

  S.clear = function () { S.n = 0; springs.clear(); grabs.clear(); held.fill(0); };

  S.spawnBlob = function (cx, cy, R, texOff) {
    const s = S.spacing, dy = s * Math.sqrt(3) / 2;
    for (let row = -Math.ceil(R / dy); row <= Math.ceil(R / dy); row++) {
      const y = row * dy, off = (row & 1) ? s / 2 : 0;
      for (let x = -R - off; x <= R; x += s) {
        const xx = x + off;
        if (xx * xx + y * y > R * R || S.n >= MAXN) continue;
        const i = S.n++;
        px[i] = cx + xx + (Math.random() - 0.5) * 0.02 * s;
        py[i] = cy + y + (Math.random() - 0.5) * 0.02 * s;
        vx[i] = vy[i] = 0; held[i] = 0;
        tu[i] = texOff + (xx + R) * 0.9; tv[i] = texOff + (y + R) * 0.9;
        spark[i] = Math.random() < 0.14 ? 0.5 + Math.random() * 6 : 0;
      }
    }
  };

  S.resize = function (w, hh) { S.W = w; S.H = hh; };

  // ---- hands ----
  S.addGrab = function (id, x, y, r, pin) {
    const idx = [], ox = [], oy = [];
    for (let i = 0; i < S.n; i++) {
      if (held[i]) continue;
      const dx = px[i] - x, dy = py[i] - y;
      if (dx * dx + dy * dy < r * r) { idx.push(i); ox.push(dx * 0.85); oy.push(dy * 0.85); held[i] = 1; }
    }
    if (!idx.length) return 0;
    grabs.set(id, { x, y, pin: !!pin, idx, ox, oy });
    return idx.length;
  };
  S.moveGrab = function (id, x, y) { const g = grabs.get(id); if (g) { g.x = x; g.y = y; } };
  S.removeGrab = function (id) {
    const g = grabs.get(id); if (!g) return;
    for (const i of g.idx) held[i] = 0;
    grabs.delete(id);
  };

  function buildPairs() {
    const n = S.n, cs = h;
    const cols = Math.ceil(S.W / cs) + 2, rows = Math.ceil(S.H / cs) + 2, nc = cols * rows;
    if (cellCount.length < nc) { cellCount = new Int32Array(nc); cellStart = new Int32Array(nc + 1); }
    cellCount.fill(0, 0, nc);
    for (let i = 0; i < n; i++) {
      let cx = (px[i] / cs) | 0, cy = (py[i] / cs) | 0;
      cx = cx < 0 ? 0 : cx >= cols ? cols - 1 : cx;
      cy = cy < 0 ? 0 : cy >= rows ? rows - 1 : cy;
      const c = cy * cols + cx; cellOf[i] = c; cellCount[c]++;
    }
    cellStart[0] = 0;
    for (let c = 0; c < nc; c++) cellStart[c + 1] = cellStart[c] + cellCount[c];
    cellCount.fill(0, 0, nc);
    for (let i = 0; i < n; i++) { const c = cellOf[i]; sorted[cellStart[c] + cellCount[c]++] = i; }
    const h2 = h * h; let np = 0;
    for (let i = 0; i < n; i++) {
      const c = cellOf[i], cx = c % cols, cy = (c / cols) | 0;
      for (let yy = cy - 1; yy <= cy + 1; yy++) {
        if (yy < 0 || yy >= rows) continue;
        for (let xx = cx - 1; xx <= cx + 1; xx++) {
          if (xx < 0 || xx >= cols) continue;
          const cc = yy * cols + xx;
          for (let k = cellStart[cc], e = cellStart[cc + 1]; k < e; k++) {
            const j = sorted[k];
            if (j <= i) continue;
            const dx = px[j] - px[i], dy = py[j] - py[i];
            if (dx * dx + dy * dy < h2 && np < PAIRCAP) { pairI[np] = i; pairJ[np] = j; np++; }
          }
        }
      }
    }
    S.np = np;
  }

  S.step = function () {
    const n = S.n, W = S.W, H = S.H;

    // 1. hands, friction, gathering
    for (const g of grabs.values()) {
      for (let k = 0; k < g.idx.length; k++) {
        const i = g.idx[k];
        const tx = (g.x + g.ox[k] - px[i]) * 0.3, ty = (g.y + g.oy[k] - py[i]) * 0.3;
        vx[i] += (tx - vx[i]) * 0.6; vy[i] += (ty - vy[i]) * 0.6;
      }
    }
    const fr = S.FRICTION, mv = S.MAXV, mv2 = mv * mv;
    const gk = S.gather > 0 ? 0.05 : 0, gcx = W / 2, gcy = H / 2;
    if (S.gather > 0) S.gather--;
    for (let i = 0; i < n; i++) {
      if (!held[i]) { vx[i] *= fr; vy[i] *= fr; }
      if (gk) {   // steer gently towards the middle, slowing down on arrival
        const dx = gcx - px[i], dy = gcy - py[i], d = Math.hypot(dx, dy) + 1e-6;
        const sp = Math.min(d * 0.012, 0.08 * h) / d;
        vx[i] += (dx * sp - vx[i]) * gk; vy[i] += (dy * sp - vy[i]) * gk;
      }
      const v2 = vx[i] * vx[i] + vy[i] * vy[i];
      if (v2 > mv2) { const s = mv / Math.sqrt(v2); vx[i] *= s; vy[i] *= s; }
    }

    buildPairs();
    const np = S.np, invH = 1 / h;

    // 2. viscosity (pairwise velocity impulses)
    const sig = S.SIGMA, beta = S.BETA;
    for (let p = 0; p < np; p++) {
      const i = pairI[p], j = pairJ[p];
      const dx = px[j] - px[i], dy = py[j] - py[i];
      const r = Math.sqrt(dx * dx + dy * dy); if (r < 1e-6) continue;
      const ux = dx / r, uy = dy / r, q = r * invH;
      const u = (vx[i] - vx[j]) * ux + (vy[i] - vy[j]) * uy;
      const I = (1 - q) * (sig * u + (u > 0 ? beta * u * u : 0)) * 0.5;
      vx[i] -= I * ux; vy[i] -= I * uy; vx[j] += I * ux; vy[j] += I * uy;
    }

    // 3. predict positions
    for (let i = 0; i < n; i++) { qx[i] = px[i]; qy[i] = py[i]; px[i] += vx[i]; py[i] += vy[i]; }

    // 4. plastic springs
    for (let p = 0; p < np; p++) {
      const key = (pairI[p] << 12) | pairJ[p];
      if (!springs.has(key)) {
        const i = pairI[p], j = pairJ[p];
        const r = Math.hypot(px[j] - px[i], py[j] - py[i]);
        if (r < h) springs.set(key, r);
      }
    }
    const ks = S.KS, gam = S.GAMMA, alp = S.ALPHA;
    for (const [key, rest0] of springs) {
      const i = key >>> 12, j = key & 4095;
      if (i >= n || j >= n) { springs.delete(key); continue; }
      const dx = px[j] - px[i], dy = py[j] - py[i];
      const L = Math.sqrt(dx * dx + dy * dy);
      if (L > 2.5 * h || L < 1e-6) { springs.delete(key); continue; }
      let rest = rest0; const d = gam * rest;
      if (L > rest + d) rest += alp * (L - rest - d);
      else if (L < rest - d) rest -= alp * (rest - d - L);
      if (rest > S.BREAK) { springs.delete(key); continue; }
      springs.set(key, rest);
      const D = ks * (1 - rest * invH) * (rest - L) / L * 0.5;
      px[i] -= D * dx; py[i] -= D * dy; px[j] += D * dx; py[j] += D * dy;
    }

    // 5. double density relaxation
    rho.fill(0, 0, n); rhoN.fill(0, 0, n); ddx.fill(0, 0, n); ddy.fill(0, 0, n);
    for (let p = 0; p < np; p++) {
      const i = pairI[p], j = pairJ[p];
      const dx = px[j] - px[i], dy = py[j] - py[i];
      const q = Math.sqrt(dx * dx + dy * dy) * invH;
      if (q >= 1) continue;
      const a = 1 - q, a2 = a * a, a3 = a2 * a;
      rho[i] += a2; rho[j] += a2; rhoN[i] += a3; rhoN[j] += a3;
    }
    const K = S.K, KN = S.KN, R0 = S.R0;
    for (let p = 0; p < np; p++) {
      const i = pairI[p], j = pairJ[p];
      const dx = px[j] - px[i], dy = py[j] - py[i];
      const r = Math.sqrt(dx * dx + dy * dy), q = r * invH;
      if (q >= 1 || r < 1e-6) continue;
      const a = 1 - q;
      const P = K * (rho[i] + rho[j] - 2 * R0) * 0.5, PN = KN * (rhoN[i] + rhoN[j]) * 0.5;
      const D = (P * a + PN * a * a) * h * 0.25 / r;
      ddx[i] -= D * dx; ddy[i] -= D * dy; ddx[j] += D * dx; ddy[j] += D * dy;
    }
    for (let i = 0; i < n; i++) { px[i] += ddx[i]; py[i] += ddy[i]; }

    // 6. fingers poking empty space push slime away
    for (const pk of pokers.values()) {
      const r2 = pk.r * pk.r;
      for (let i = 0; i < n; i++) {
        const dx = px[i] - pk.x, dy = py[i] - pk.y, d2 = dx * dx + dy * dy;
        if (d2 < r2 && d2 > 1e-6) { const d = Math.sqrt(d2), s = (pk.r - d) / d * 0.5; px[i] += dx * s; py[i] += dy * s; }
      }
    }

    // 7. walls, then velocity from displacement
    const m = 0.3 * h;
    for (let i = 0; i < n; i++) {
      if (px[i] < m) px[i] = m; else if (px[i] > W - m) px[i] = W - m;
      if (py[i] < m) py[i] = m; else if (py[i] > H - m) py[i] = H - m;
      vx[i] = px[i] - qx[i]; vy[i] = py[i] - qy[i];
    }
  };

  // Count connected pieces with at least minSize particles.
  S.countBlobs = function (minSize) {
    const n = S.n;
    for (let i = 0; i < n; i++) { uf[i] = i; ufSize[i] = 0; }
    const find = (a) => { while (uf[a] !== a) { uf[a] = uf[uf[a]]; a = uf[a]; } return a; };

    for (let p = 0; p < S.np; p++) {

    }
    for (let i = 0; i < n; i++) ufSize[find(i)]++;
    let c = 0;
    for (let i = 0; i < n; i++) if (ufSize[i] >= minSize) c++;
    return c;
  };

  return S;
}

      const wx = x + 160 * (nB(x, y) - 0.5), wy = y + 160 * (nC(x, y) - 0.5);
      let t = nA(wx, wy);
      t = Math.min(1, Math.max(0, (t - 0.27) / 0.48));
      t = t * t * (3 - 2 * t);
      const f = t * (stops.length - 1), k = Math.min(stops.length - 2, f | 0), u = f - k;
      let r = stops[k][0] + (stops[k + 1][0] - stops[k][0]) * u;
      let g = stops[k][1] + (stops[k + 1][1] - stops[k][1]) * u;
      let b = stops[k][2] + (stops[k + 1][2] - stops[k][2]) * u;
      let a = nD(wy, wx); a = Math.max(0, (a - 0.55) / 0.2); a = Math.min(1, a) * 0.55;
      r += (acc[0] - r) * a; g += (acc[1] - g) * a; b += (acc[2] - b) * a;
      const o = (y * TEX + x) * 3; out[o] = r; out[o + 1] = g; out[o + 2] = b;
    }
    // stars
    function star(cx, cy, rad, br, tint) {
      const R = Math.ceil(rad * 3);
      for (let dy = -R; dy <= R; dy++) for (let dx = -R; dx <= R; dx++) {
        const d2 = (dx * dx + dy * dy) / (rad * rad);
        let w = Math.exp(-d2) * br;
        if (rad > 1.5) w += br * 0.5 * Math.exp(-Math.abs(dy) / 0.7 - Math.abs(dx) / (rad * 2.5)) * (dy === 0 ? 1 : 0.2)
                         + br * 0.5 * Math.exp(-Math.abs(dx) / 0.7 - Math.abs(dy) / (rad * 2.5)) * (dx === 0 ? 1 : 0.2);
        if (w < 0.01) continue;
        const o = ((((cy + dy) & TMASK) * TEX) + ((cx + dx) & TMASK)) * 3;
        out[o] += w * tint[0]; out[o + 1] += w * tint[1]; out[o + 2] += w * tint[2];
      }
    }
    for (let i = 0; i < 1500; i++) {
      const tint = rnd() < 0.3 ? [180, 210, 255] : rnd() < 0.5 ? [255, 230, 200] : [255, 255, 255];
      star((rnd() * TEX) | 0, (rnd() * TEX) | 0, 0.5 + rnd() * 0.7, 0.25 + rnd() * 0.75, tint);
    }
    for (let i = 0; i < 22; i++) star((rnd() * TEX) | 0, (rnd() * TEX) | 0, 1.6 + rnd() * 1.4, 0.8, [255, 245, 255]);
    const t8 = new Uint8ClampedArray(out.length);
    for (let i = 0; i < out.length; i++) t8[i] = out[i];
    return t8;

  function renderField(time) {
    const n = sim.n, px = sim.px, py = sim.py, tu = sim.tu, tv = sim.tv;
    field.fill(0); fu.fill(0); fv.fill(0);
    const rc = KR / FS;
    for (let i = 0; i < n; i++) {
      const cx = px[i] / FS - 0.5, cy = py[i] / FS - 0.5, u = tu[i], v = tv[i];
      const x0 = Math.max(0, Math.ceil(cx - rc)), x1 = Math.min(fw - 1, Math.floor(cx + rc));
      const y0 = Math.max(0, Math.ceil(cy - rc)), y1 = Math.min(fh - 1, Math.floor(cy + rc));
      for (let y = y0; y <= y1; y++) {
        const dy = (y - cy) * FS, dy2 = dy * dy, row = y * fw;
        for (let x = x0; x <= x1; x++) {
          const dx = (x - cx) * FS, d2 = dx * dx + dy2;
          if (d2 >= KR2) continue;
          const t = 1 - d2 / KR2, w = t * t * t, k = row + x;
          field[k] += w; fu[k] += w * u; fv[k] += w * v;
        }
      }
    }
    const data = img.data, tx = tex, rr = rimCol[0], rg = rimCol[1], rb = rimCol[2];
    const driftU = time * 0.004, driftV = time * 0.0025;
    const sCoef = ZH * invT / (2 * FS);
    data.fill(0);
    for (let y = 1; y < fh - 1; y++) {
      for (let x = 1; x < fw - 1; x++) {
        const k = y * fw + x, f = field[k];
        if (f < T * 0.72) continue;
        let a = (f - T * 0.72) / (T * 0.4); if (a > 1) a = 1;
        const inv = 1 / f;
        const ti = ((((fv[k] * inv + driftV) | 0) & TMASK) * TEX + (((fu[k] * inv + driftU) | 0) & TMASK)) * 3;
        const fe = f < T ? T : f;
        const s = sCoef * Math.exp(-(fe - T) * invT * 0.9);
        const nx = -(field[k + 1] - field[k - 1]) * s, ny = -(field[k + fw] - field[k - fw]) * s;
        const nl = 1 / Math.sqrt(nx * nx + ny * ny + 1);
        const ux = nx * nl, uy = ny * nl, uz = nl;
        let dif = ux * LX + uy * LY + uz * LZ; if (dif < 0) dif = 0;
        let sp = ux * HX + uy * HY + uz * HZ; sp = sp > 0 ? sp * sp : 0; sp *= sp; sp *= sp; sp *= sp; sp *= sp; // ^32
        const rim = (1 - uz) * (1 - uz);
        const lit = 0.5 + 0.7 * dif, spec = sp * 230;
        const o = k * 4;
        data[o]     = tx[ti] * lit + spec + rr * rim * 0.55;
        data[o + 1] = tx[ti + 1] * lit + spec + rg * rim * 0.55;
        data[o + 2] = tx[ti + 2] * lit + spec + rb * rim * 0.55;
        data[o + 3] = a * 255;
      }
    }
    fctx.putImageData(img, 0, 0);
  }

  // ---------- sound ----------
  let ac = null, soundOn = true, noiseBuf = null;
